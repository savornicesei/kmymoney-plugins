cmake_minimum_required(VERSION 3.21)
project(KMyMoneyCommunityPlugins VERSION 0.1.0 LANGUAGES CXX)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH "${ECM_MODULE_PATH}" "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(KDEInstallDirs6)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets)
find_package(KF6 REQUIRED COMPONENTS CoreAddons I18n XmlGui)
find_package(LibAlkimia6 REQUIRED)
find_package(KMyMoneyHost REQUIRED)
include(CTest)

set(KMM_PLUGIN_SELECTION "all" CACHE STRING "Plugin directories to include (semicolon-separated), or all")
file(GLOB plugin_projects RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/plugins"
    CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/*/CMakeLists.txt")
set(available_plugins)
foreach(plugin_project IN LISTS plugin_projects)
    get_filename_component(plugin "${plugin_project}" DIRECTORY)
    list(APPEND available_plugins "${plugin}")
endforeach()
if(NOT KMM_PLUGIN_SELECTION STREQUAL "all")
    foreach(plugin IN LISTS KMM_PLUGIN_SELECTION)
        if(NOT plugin IN_LIST available_plugins)
            message(FATAL_ERROR "Unknown plugin '${plugin}'. Available: ${available_plugins}")
        endif()
    endforeach()
endif()
foreach(plugin IN LISTS available_plugins)
    string(TOUPPER "BUILD_${plugin}" plugin_option)
    string(REPLACE "-" "_" plugin_option "${plugin_option}")
    option(${plugin_option} "Build the ${plugin} plugin" ON)
    if(${plugin_option} AND (KMM_PLUGIN_SELECTION STREQUAL "all" OR plugin IN_LIST KMM_PLUGIN_SELECTION))
        add_subdirectory("plugins/${plugin}")
    endif()
endforeach()
